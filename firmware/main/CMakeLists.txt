idf_component_register(SRCS "esptelepathy_main.c" INCLUDE_DIRS "." REQUIRES esp_app_format)

execute_process(
    COMMAND git rev-parse --short=12 HEAD
    WORKING_DIRECTORY "${CMAKE_CURRENT_LIST_DIR}/../.."
    RESULT_VARIABLE ESPTELEPATHY_GIT_RESULT
    OUTPUT_VARIABLE ESPTELEPATHY_GIT_SHA
    OUTPUT_STRIP_TRAILING_WHITESPACE
    ERROR_QUIET
)

if(NOT ESPTELEPATHY_GIT_RESULT EQUAL 0 OR ESPTELEPATHY_GIT_SHA STREQUAL "")
    set(ESPTELEPATHY_GIT_SHA "unknown")
endif()

target_compile_definitions(
    ${COMPONENT_LIB}
    PRIVATE
        ESPTELEPATHY_GIT_SHA="${ESPTELEPATHY_GIT_SHA}"
        ESPTELEPATHY_BUILD_TARGET="${IDF_TARGET}"
)
